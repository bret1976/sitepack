  const h = String(Math.floor(d/3600000)).padStart(2,'0');
  const m = String(Math.floor(d/60000)%60).padStart(2,'0');
  const s = String(Math.floor(d/1000)%60).padStart(2,'0');
  document.getElementById('timer').textContent = `${h}:${m}:${s}`;
}
setInterval(tick, 1000); tick();

const container = document.getElementById('looks');

    let y = 12;

      d += ` L ${x.toFixed(1)} ${y.toFixed(1)}`;
    }
    paths += `<path d="${d}" stroke="${lk.hue}" stroke-opacity="${0.75 - s*0.22}" stroke-width="${1.1 - s*0.3}" fill="none"/>`;

  // horizontal scan slices
