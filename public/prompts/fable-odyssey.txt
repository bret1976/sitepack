const cv = document.getElementById('space');
const g = cv.getContext('2d');
let W, H, DPR;
function rs(){
  DPR = Math.min(devicePixelRatio, 2);
  W = cv.width = innerWidth*DPR; H = cv.height = innerHeight*DPR;
  cv.style.width = innerWidth+'px'; cv.style.height = innerHeight+'px';
}
addEventListener('resize', rs); rs();

// starfield layers
const stars = Array.from({length:420}, () => ({
  x:Math.random(), y:Math.random(), z:0.3+Math.random()*0.7, tw:Math.random()*7
}));

// bodies keyed by section
const BODIES = {
  mercury: { r:26, col:['#b8a89a','#8a7a6c','#5a4f45'], ring:false },
  venus:   { r:40, col:['#f0d9a8','#e0b97a','#c99a58'], ring:false },
  earth:   { r:42, col:['#7fb4ff','#3a7bd5','#1d4e89'], ring:false, earth:true },
  mars:    { r:34, col:['#e88a5c','#c1552f','#8a3a1e'], ring:false },
  jupiter: { r:78, col:['#e8c9a0','#c9975c','#a06a3a'], ring:false, bands:true },
  saturn:  { r:62, col:['#f0dcae','#d9bc82','#b3955e'], ring:true },
  neptune: { r:44, col:['#8ab4f0','#4a72c4','#2a4a8a'], ring:false },
};

const sections = [...document.querySelectorAll('section[data-body]')];
const cards = [...document.querySelectorAll('.card')];
const io = new IntersectionObserver(es => es.forEach(e => {
  const card = e.target.querySelector('.card');
  if (e.isIntersecting) card.classList.add('in');
}), { threshold:0.35 });
sections.forEach(s => io.observe(s));

const distEl = document.getElementById('dist');
const railFill = document.getElementById('railFill');

function draw(now){
  const t = now/1000;
  g.clearRect(0,0,W,H);
  const total = document.body.scrollHeight - innerHeight;
  const p = scrollY/Math.max(1,total);

  // distance HUD: 0 -> 4.5B km, eased
  const km = Math.pow(p, 1.6) * 4.5e9;
  distEl.textContent = km < 1e6
    ? Math.round(km).toLocaleString() + ' km from the Sun'
    : (km/1e6).toFixed(1) + ' million km from the Sun';
  railFill.style.height = (p*100) + '%';

  // sun recedes: big glow at top-left early, shrinks with p
  const sunR = Math.max(8, (1-p)*(1-p) * H*0.5);
  const sx = W*0.5 - p*W*0.2, sy = -H*0.12 - p*H*0.4;
  const sg = g.createRadialGradient(sx, sy, 0, sx, sy, sunR*2.4);
  sg.addColorStop(0, 'rgba(255,240,200,0.9)');
  sg.addColorStop(0.35, 'rgba(255,190,90,0.35)');
  sg.addColorStop(1, 'transparent');
  g.fillStyle = sg;
  g.fillRect(0,0,W,H);

  // stars with parallax drift as you scroll
  for (const s of stars){
    const y = ((s.y + p * s.z * 1.6) % 1.05) * H;
    const x = s.x * W;
    const tw = 0.5 + 0.5*Math.sin(t*1.5 + s.tw*9);
    g.fillStyle = `rgba(220,230,255,${(0.14 + 0.5*s.z) * (0.6+0.4*tw)})`;
    g.beginPath();
    g.arc(x, y, s.z * 1.5 * DPR, 0, 7);
    g.fill();
  }

  // planets: each rendered while its section is near viewport
  sections.forEach(sec => {
    const r = sec.getBoundingClientRect();
    const mid = r.top + r.height/2;
    const vis = 1 - Math.min(1, Math.abs(mid - innerHeight/2) / (innerHeight*0.95));
    if (vis <= 0.02) return;
    const b = BODIES[sec.dataset.body];
    const right = !sec.classList.contains('right');
    const px = (right ? W*0.68 : W*0.32) + (right?1:-1) * (1-vis) * W*0.1;
    const py = H*0.5 + (mid - innerHeight/2) * DPR * 0.55;
    const R = b.r * DPR * (1.25 + vis*0.45) * Math.min(1, W/(1300*DPR)+0.55);

    g.save();
    g.globalAlpha = Math.min(1, vis*1.6);

    // ring behind
    if (b.ring){
      g.save();
      g.translate(px, py);
      g.rotate(-0.42);
      g.scale(1, 0.32);
      g.strokeStyle = 'rgba(220,205,160,0.55)';
      g.lineWidth = R*0.34;
      g.beginPath(); g.arc(0, 0, R*1.75, Math.PI*0.04, Math.PI*1.04, false); g.stroke();
      g.restore();
    }

    // sphere
    const pg = g.createRadialGradient(px - R*0.4, py - R*0.42, R*0.1, px, py, R);
    pg.addColorStop(0, b.col[0]);
    pg.addColorStop(0.55, b.col[1]);
    pg.addColorStop(1, b.col[2]);
    g.fillStyle = pg;
    g.beginPath(); g.arc(px, py, R, 0, 7); g.fill();

    // jupiter bands
    if (b.bands){
      g.save();
      g.beginPath(); g.arc(px, py, R, 0, 7); g.clip();
      for (let i=-3;i<=3;i++){
        g.fillStyle = i%2 ? 'rgba(160,106,58,0.35)' : 'rgba(240,220,190,0.22)';
        g.fillRect(px-R, py + i*R*0.26 - R*0.09 + Math.sin(t*0.4+i)*R*0.02, R*2, R*0.17);
      }
      // great red spot
      g.fillStyle = 'rgba(200,80,50,0.75)';
      g.beginPath();
      g.ellipse(px+R*0.34, py+R*0.3, R*0.2, R*0.12, 0.2, 0, 7);
      g.fill();
      g.restore();
    }
    // earth clouds+city glint
    if (b.earth){
      g.save();
      g.beginPath(); g.arc(px, py, R, 0, 7); g.clip();
      g.fillStyle = 'rgba(90,200,120,0.5)';
      g.beginPath(); g.ellipse(px-R*0.25, py-R*0.1, R*0.38, R*0.26, 0.6, 0, 7); g.fill();
      g.beginPath(); g.ellipse(px+R*0.4, py+R*0.35, R*0.26, R*0.18, -0.4, 0, 7); g.fill();
      g.fillStyle = 'rgba(255,255,255,0.55)';
      for (let i=0;i<5;i++){
        g.beginPath();
        g.ellipse(px + Math.sin(t*0.1+i*2)*R*0.7, py - R*0.5 + i*R*0.26, R*0.34, R*0.06, 0.15, 0, 7);
        g.fill();
      }
      g.restore();
    }

    // terminator shadow
    const sh = g.createRadialGradient(px + R*0.65, py + R*0.2, R*0.2, px, py, R*1.35);
    sh.addColorStop(0, 'transparent');
    sh.addColorStop(0.62, 'transparent');
    sh.addColorStop(1, 'rgba(1,2,8,0.92)');
    g.fillStyle = sh;
    g.beginPath(); g.arc(px, py, R, 0, 7); g.fill();

    // ring in front
    if (b.ring){
      g.save();
      g.translate(px, py);
      g.rotate(-0.42);
      g.scale(1, 0.32);
      g.strokeStyle = 'rgba(232,218,175,0.85)';
      g.lineWidth = R*0.34;
      g.beginPath(); g.arc(0, 0, R*1.75, -Math.PI*0.96, Math.PI*0.04, false); g.stroke();
      g.strokeStyle = 'rgba(120,105,80,0.6)';
      g.lineWidth = R*0.05;
      g.beginPath(); g.arc(0, 0, R*1.52, -Math.PI, 0, false); g.stroke();
      g.restore();
    }
    g.restore();
  });

  requestAnimationFrame(draw);
}
requestAnimationFrame(draw);
</script>
