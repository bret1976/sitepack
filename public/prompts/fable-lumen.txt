// blob drift — slow lissajous orbits + subtle mouse parallax
const blobs = [...document.querySelectorAll('.blob')];
const P = [
  { cx:0.18, cy:0.22, ax:0.10, ay:0.14, sx:0.11, sy:0.09, ph:0 },
  { cx:0.72, cy:0.30, ax:0.12, ay:0.10, sx:0.07, sy:0.12, ph:2 },
  { cx:0.45, cy:0.72, ax:0.14, ay:0.09, sx:0.09, sy:0.07, ph:4 },
  { cx:0.85, cy:0.75, ax:0.08, ay:0.12, sx:0.12, sy:0.10, ph:1 },
];
let mx = 0.5, my = 0.5;
addEventListener('pointermove', e => {
  mx = e.clientX/innerWidth; my = e.clientY/innerHeight;
});
(function drift(now){
  const t = now/1000;
  blobs.forEach((b, i) => {
    const p = P[i];
    const x = (p.cx + Math.sin(t*p.sx + p.ph)*p.ax + (mx-0.5)*0.04*(i+1)) * innerWidth;
    const y = (p.cy + Math.cos(t*p.sy + p.ph)*p.ay + (my-0.5)*0.04*(i+1)) * innerHeight + scrollY*0.15*(i%2?1:-1);
    b.style.transform = `translate(${x - b.offsetWidth/2}px, ${y - b.offsetHeight/2}px) scale(${1 + Math.sin(t*0.3+p.ph)*0.12})`;
  });
  requestAnimationFrame(drift);
})(0);

const io = new IntersectionObserver(es => es.forEach(e => {
  if (e.isIntersecting) e.target.classList.add('in');
}), { threshold:0.2 });
document.querySelectorAll('.reveal').forEach((el,i) => {
