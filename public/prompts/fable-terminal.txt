function esc(s){ return s.replace(/&/g,'&amp;').replace(/</g,'&lt;'); }

function println(html, cls=''){
  const div = document.createElement('div');
  if (cls) div.className = cls;
  div.innerHTML = html || '&nbsp;';
  screen.appendChild(div);
  screen.scrollTop = screen.scrollHeight;
  return div;
}

// typewriter that types into a fresh element
function typeOut(text, cls='', speed=8){
  return new Promise(res => {
    const el = println('', cls);
    let i = 0;
    const pre = document.createElement('pre');
    el.appendChild(pre);
    (function tick(){
      const chunk = text.slice(i, i + 3);
      pre.textContent += chunk;
      i += 3;
      screen.scrollTop = screen.scrollHeight;
      if (i < text.length) setTimeout(tick, speed);
      else res();
    })();
  });
}

function prompt(){
  const line = println('');
  line.innerHTML = `<span class="amber">OBSOLETE&gt;</span> `;
  const input = document.createElement('input');
  input.className = 'inline-input';
  input.id = 'cmd';
  input.autocapitalize = 'off'; input.autocomplete = 'off'; input.spellcheck = false;
  line.appendChild(input);
  const cur = document.createElement('span');
  cur.className = 'cursor';
  line.appendChild(cur);
  input.focus();
  input.addEventListener('input', () => { /* keep cursor after text */ });
  input.addEventListener('keydown', async e => {
    if (e.key !== 'Enter') return;
    const cmd = input.value.trim().toUpperCase();
    line.innerHTML = `<span class="amber">OBSOLETE&gt;</span> <span class="white">${esc(input.value)}</span>`;
    await exec(cmd);
    prompt();
  });
  screen.scrollTop = screen.scrollHeight;
}

async function exec(cmd){
  if (!cmd) return;
  if (cmd === 'HELP') { await typeOut(HELP, 'dim', 3); return; }
  if (cmd === 'LIST') {
    await typeOut(
`CATALOG — PERMANENT COLLECTION
 -----------------------------
 [1] ENIAC ................. 1945
 [2] ALTAIR 8800 ........... 1975
 [3] CRAY-1 ................ 1976

 TYPE: VIEW <NUMBER>`, '', 3);
    return;
  }
  if (cmd.startsWith('VIEW')) {
    const n = cmd.replace('VIEW','').trim();
    const ex = EXHIBITS[n];
    if (ex) {
      println(`<span class="amber">■ EXHIBIT ${n} — ${esc(ex.name)}</span>`);
      await typeOut(ex.body, '', 6);
    } else {
      await typeOut(` NO SUCH EXHIBIT. TYPE LIST.`, 'dim', 6);
    }
    return;
  }
  if (cmd === 'ABOUT') {
    await typeOut(
` THE MUSEUM OF DEAD MACHINES OPERATES THIS

 WAS DECOMMISSIONED IN 1989; THE COLLECTION

    return;

  await typeOut(` ?SYNTAX ERROR: "${cmd}". TYPE HELP.`, 'dim', 6);
}
