      const scene = document.getElementById('heroScene');

      const canvas = document.querySelector('.background-canvas');
      const context = canvas.getContext('2d');

      const cursor = document.querySelector('.luxury-cursor');
      const cursorDot = document.querySelector('.luxury-cursor-dot');
      const magneticItems = [...document.querySelectorAll('.lux-button, .destination, .panel-link, .nav-links a, .nav-pill')];
      const clamp = (value, min, max) => Math.min(Math.max(value, min), max);
      const frameCount = 241;
      let activeFrame = 1;
      let targetFrame = 1;
      let latestProgress = 0;
      let mouseX = window.innerWidth / 2;
      let mouseY = window.innerHeight / 2;
      let cursorX = mouseX;
      let cursorY = mouseY;
      let cursorDotX = mouseX;
      let cursorDotY = mouseY;
      const frames = new Array(frameCount + 1);
      const frameSrc = (index) => `hero-frames/frame-${String(index).padStart(3, '0')}.jpg`;
      const dust = Array.from({ length: 54 }, (_, index) => {
        const seed = Math.sin(index * 117.33) * 10000;
        const random = seed - Math.floor(seed);
        const seedTwo = Math.sin(index * 41.71) * 10000;
        const randomTwo = seedTwo - Math.floor(seedTwo);
        return {
          x: random,
          y: randomTwo,
          size: .7 + ((index * 13) % 19) / 10,
          depth: .18 + ((index * 7) % 17) / 17,
          opacity: .08 + ((index * 5) % 12) / 100
        };
      });

      function drawCover(image) {
        const width = canvas.clientWidth;
        const height = canvas.clientHeight;
        const ratio = window.devicePixelRatio || 1;

        if (canvas.width !== Math.round(width * ratio) || canvas.height !== Math.round(height * ratio)) {
          canvas.width = Math.round(width * ratio);
          canvas.height = Math.round(height * ratio);
        }

        const canvasRatio = canvas.width / canvas.height;
        const imageRatio = image.naturalWidth / image.naturalHeight;
        let sourceWidth = image.naturalWidth;
        let sourceHeight = image.naturalHeight;
        let sourceX = 0;
        let sourceY = 0;

        if (imageRatio > canvasRatio) {
          sourceWidth = image.naturalHeight * canvasRatio;
          sourceX = (image.naturalWidth - sourceWidth) / 2;
        } else {
          sourceHeight = image.naturalWidth / canvasRatio;
          sourceY = (image.naturalHeight - sourceHeight) / 2;
        }

        context.clearRect(0, 0, canvas.width, canvas.height);
        context.drawImage(image, sourceX, sourceY, sourceWidth, sourceHeight, 0, 0, canvas.width, canvas.height);
      }

      function nearestLoadedFrame(index) {
        if (frames[index]?.complete) return frames[index];
        for (let offset = 1; offset < frameCount; offset += 1) {
          const lower = index - offset;
          const upper = index + offset;
          if (lower >= 1 && frames[lower]?.complete) return frames[lower];
          if (upper <= frameCount && frames[upper]?.complete) return frames[upper];
        }
        return null;
      }

      function renderTargetFrame() {
        const image = nearestLoadedFrame(targetFrame);
        if (!image) return;
        drawCover(image);
        activeFrame = targetFrame;
      }

      function loadFrame(index, highPriority = false) {
        if (frames[index]) return frames[index];
        const image = new Image();
        image.decoding = 'async';
        if (highPriority) image.fetchPriority = 'high';
        image.onload = () => {
          if (Math.abs(index - targetFrame) <= 1 || index === 1) renderTargetFrame();
        };
        image.src = frameSrc(index);
        frames[index] = image;
        return image;
      }

      function preloadFramesAround(index) {
        for (let offset = -8; offset <= 8; offset += 1) {
          const candidate = index + offset;
          if (candidate >= 1 && candidate <= frameCount) loadFrame(candidate, Math.abs(offset) <= 2);
        }
      }

      function preloadAllFrames() {
        let index = 1;
        const loadBatch = () => {
          const end = Math.min(frameCount, index + 11);
          for (; index <= end; index += 1) loadFrame(index);
          if (index <= frameCount) window.setTimeout(loadBatch, 24);
        }
        loadBatch();
      }

      function resizeDepthCanvas() {
        const ratio = window.devicePixelRatio || 1;
        const width = depthCanvas.clientWidth;
        const height = depthCanvas.clientHeight;
        if (depthCanvas.width !== Math.round(width * ratio) || depthCanvas.height !== Math.round(height * ratio)) {
          depthCanvas.width = Math.round(width * ratio);
          depthCanvas.height = Math.round(height * ratio);
        }
      }

      function drawDepthParticles(progress) {
        resizeDepthCanvas();
        const width = depthCanvas.width;
        const height = depthCanvas.height;
        depthContext.clearRect(0, 0, width, height);
        dust.forEach((particle, index) => {
          const drift = progress * width * (.12 + particle.depth * .26);
          const lift = progress * height * (.04 + particle.depth * .08);
          const wave = Math.sin(progress * Math.PI * 2 + index) * 24 * particle.depth;
          const x = ((particle.x * width + drift + wave) % (width + 80)) - 40;
          const y = ((particle.y * height - lift + height * .18) % (height + 80)) - 40;
          const radius = particle.size * (window.devicePixelRatio || 1);
          const glow = depthContext.createRadialGradient(x, y, 0, x, y, radius * 7);
          glow.addColorStop(0, `rgba(255, 214, 150, ${particle.opacity})`);
          glow.addColorStop(.42, `rgba(214, 156, 80, ${particle.opacity * .34})`);
          glow.addColorStop(1, 'rgba(214, 156, 80, 0)');
          depthContext.fillStyle = glow;
          depthContext.beginPath();
          depthContext.arc(x, y, radius * 7, 0, Math.PI * 2);
          depthContext.fill();
        });
      }

      function setScrollProgress() {
        const maxScroll = Math.max(1, document.documentElement.scrollHeight - window.innerHeight);
        const pageProgress = clamp(window.scrollY / maxScroll, 0, 1);
        const heroProgress = clamp(window.scrollY / Math.max(1, window.innerHeight * .95), 0, 1);

        root.style.setProperty('--progress', pageProgress.toFixed(4));
        root.style.setProperty('--hero-progress', heroProgress.toFixed(4));

        sections.forEach((section) => {
          const box = section.getBoundingClientRect();
          const span = Math.max(1, window.innerHeight + box.height);
          const raw = clamp((window.innerHeight - box.top) / span, 0, 1);
          const enter = Math.sin(raw * Math.PI);
          section.style.setProperty('--section-progress', raw.toFixed(4));
          section.style.setProperty('--section-enter', enter.toFixed(4));
        });

        latestProgress = pageProgress;
        targetFrame = Math.round(1 + latestProgress * (frameCount - 1));
        if (targetFrame !== activeFrame) {
          loadFrame(targetFrame, true);
          preloadFramesAround(targetFrame);
          renderTargetFrame();
        }

        drawDepthParticles(pageProgress);
      }

      setScrollProgress();
      window.addEventListener('scroll', setScrollProgress, { passive: true });
      window.addEventListener('resize', setScrollProgress);

      loadFrame(1, true);
      preloadFramesAround(activeFrame);
      const idlePreload = window.requestIdleCallback || ((callback) => window.setTimeout(callback, 800));
      idlePreload(preloadAllFrames);

      if (window.matchMedia('(hover: hover) and (pointer: fine)').matches) {
        window.addEventListener('mousemove', (event) => {
          mouseX = event.clientX;
          mouseY = event.clientY;
          document.body.classList.add('cursor-ready');
        }, { passive: true });

        magneticItems.forEach((item) => {
          item.addEventListener('mousemove', (event) => {
            const rect = item.getBoundingClientRect();
            const x = event.clientX - (rect.left + rect.width / 2);
            const y = event.clientY - (rect.top + rect.height / 2);
            item.style.transform = `translate3d(${x * .08}px, ${y * .12}px, 0)`;
            document.body.classList.add('cursor-active');
          });

          item.addEventListener('mouseleave', () => {
            item.style.transform = '';
            document.body.classList.remove('cursor-active');
          });
        });

        function animateCursor() {
          cursorX += (mouseX - cursorX) * .16;
          cursorY += (mouseY - cursorY) * .16;
          cursorDotX += (mouseX - cursorDotX) * .42;
          cursorDotY += (mouseY - cursorDotY) * .42;
          cursor.style.transform = `translate3d(${cursorX}px, ${cursorY}px, 0) translate(-50%, -50%)`;
          cursorDot.style.transform = `translate3d(${cursorDotX}px, ${cursorDotY}px, 0) translate(-50%, -50%)`;
          window.requestAnimationFrame(animateCursor);
        }

        animateCursor();
      }
    })();
  </script>
</body>
</html>
