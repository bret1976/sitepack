const asmTrack = document.getElementById('asmTrack');
const copies = [...document.querySelectorAll('.asm-copy')];
const easeIO = t => t<0.5 ? 4*t*t*t : 1-Math.pow(-2*t+2,3)/2;
const clamp01 = v => Math.max(0, Math.min(1, v));

function asmUpdate(){
  const r = asmTrack.getBoundingClientRect();
  const total = r.height - innerHeight;
  const p = clamp01(-r.top / total);

  // 4 phases, each 22% with gaps
  const ph = i => easeIO(clamp01((p - i*0.24) / 0.2));
  const pMove = ph(0), pDial = ph(1), pHands = ph(2), pCase = ph(3);

  // movement flies in from deep below + scales
  gMove.setAttribute('transform',
    `translate(0 ${ (1-pMove)*520 }) scale(${0.6+0.4*pMove})`);
  gMove.setAttribute('transform-origin','200 200');
  gMove.style.opacity = pMove;

  gDial.setAttribute('transform', `translate(${(1-pDial)*-460} ${(1-pDial)*-160}) rotate(${(1-pDial)*-40} 200 200)`);
  gDial.style.opacity = pDial;

  gHands.setAttribute('transform', `translate(${(1-pHands)*430} ${(1-pHands)*-240}) rotate(${(1-pHands)*70} 200 200)`);
  gHands.style.opacity = pHands;

  gCase.setAttribute('transform', `scale(${1 + (1-pCase)*0.55})`);
  gCase.setAttribute('transform-origin','200 200');
  gCase.style.opacity = pCase;

  // copy visibility
  const idx = Math.min(3, Math.floor(p/0.24));
  copies.forEach(c => c.classList.toggle('on', +c.dataset.step === idx));
}
addEventListener('scroll', asmUpdate, { passive:true }); asmUpdate();
