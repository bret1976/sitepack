<script>
// ---------------- Web Audio radio ----------------
let ctx, master, noiseGain, toneGain, noiseSrc, current = null;
let powered = false;
let freq = 90.4;      // 88..108
let volume = 0.6;

const STATIONS = [
  { at:89.1, w:0.45, name:'KLOW — SLOW WAVES', build: slowWaves },
  { at:94.7, w:0.45, name:'WDRM — NIGHT DRONE', build: nightDrone },
  { at:100.3, w:0.45, name:'KBEL — BELL TOWER', build: bellTower },
  { at:105.9, w:0.45, name:'WPLS — PULSE FM', build: pulseFM },
];

function makeNoise(){
  const len = 2 * ctx.sampleRate;
  const buf = ctx.createBuffer(1, len, ctx.sampleRate);
  const d = buf.getChannelData(0);
  for (let i = 0; i < len; i++) d[i] = Math.random()*2 - 1;
  const src = ctx.createBufferSource();
  src.buffer = buf; src.loop = true;
  return src;
}

// --- station generators: each returns {nodes:[], gain} connected to toneGain
function slowWaves(){
  const g = ctx.createGain(); g.gain.value = 0;
  const o1 = ctx.createOscillator(); o1.type='sine'; o1.frequency.value = 174;
  const o2 = ctx.createOscillator(); o2.type='sine'; o2.frequency.value = 174*1.5 + 1.2;
  const lfo = ctx.createOscillator(); lfo.frequency.value = 0.09;
  const lfoG = ctx.createGain(); lfoG.gain.value = 30;
  lfo.connect(lfoG); lfoG.connect(o2.frequency);
  const f = ctx.createBiquadFilter(); f.type='lowpass'; f.frequency.value=900;
  o1.connect(f); o2.connect(f); f.connect(g); g.connect(toneGain);
  [o1,o2,lfo].forEach(o=>o.start());
  return { nodes:[o1,o2,lfo], gain:g };
}
function nightDrone(){
  const g = ctx.createGain(); g.gain.value = 0;
  const os = [55, 55*2.01, 55*2.99, 55*4.02].map((fr,i) => {
    const o = ctx.createOscillator();
    o.type = i%2 ? 'sawtooth' : 'triangle';
    o.frequency.value = fr;
    const og = ctx.createGain(); og.gain.value = 0.22/(i+1);
    o.connect(og); og.connect(g);
    o.start();
    return o;
  });
  const f = ctx.createBiquadFilter(); f.type='lowpass'; f.frequency.value=420; f.Q.value=6;
  const lfo = ctx.createOscillator(); lfo.frequency.value=0.05;
  const lg = ctx.createGain(); lg.gain.value=180;
  lfo.connect(lg); lg.connect(f.frequency); lfo.start();
  g.disconnect(); g.connect(f); f.connect(toneGain);
  return { nodes:[...os, lfo], gain:g };
}
function bellTower(){
  const g = ctx.createGain(); g.gain.value = 0;
  g.connect(toneGain);
  let alive = true;
  function ring(){
    if (!alive || !powered) return;
    const base = [523, 659, 784, 880][Math.floor(Math.random()*4)];
    [1, 2.76, 5.4].forEach((h, i) => {
      const o = ctx.createOscillator(); o.type='sine';
      o.frequency.value = base*h * (1 + (Math.random()-0.5)*0.003);
      const og = ctx.createGain();
      og.gain.setValueAtTime(0.35/(i+1), ctx.currentTime);
      og.gain.exponentialRampToValueAtTime(0.0001, ctx.currentTime + 4 + i);
      o.connect(og); og.connect(g);
      o.start(); o.stop(ctx.currentTime + 5 + i);
    });
    setTimeout(ring, 1800 + Math.random()*2600);
  }
  ring();
  return { nodes:[], gain:g, kill:() => alive = false };
}
function pulseFM(){
  const g = ctx.createGain(); g.gain.value = 0;
  const o = ctx.createOscillator(); o.type='square'; o.frequency.value=110;
  const f = ctx.createBiquadFilter(); f.type='bandpass'; f.frequency.value=600; f.Q.value=8;
  const lfo = ctx.createOscillator(); lfo.type='square'; lfo.frequency.value=2.2;
  const lg = ctx.createGain(); lg.gain.value=0.5;
  const vca = ctx.createGain(); vca.gain.value=0.5;
  lfo.connect(lg); lg.connect(vca.gain);
  const sweep = ctx.createOscillator(); sweep.frequency.value=0.11;
  const sg = ctx.createGain(); sg.gain.value=420;
  sweep.connect(sg); sg.connect(f.frequency);
  o.connect(f); f.connect(vca); vca.connect(g); g.connect(toneGain);
  [o,lfo,sweep].forEach(x=>x.start());
  return { nodes:[o,lfo,sweep], gain:g };
}

function initAudio(){
  ctx = new (window.AudioContext || window.webkitAudioContext)();
  master = ctx.createGain(); master.gain.value = volume;
  master.connect(ctx.destination);
  noiseGain = ctx.createGain(); noiseGain.gain.value = 0;
  toneGain = ctx.createGain(); toneGain.gain.value = 1;
  // radio voicing: bandpass the lot
  const voice = ctx.createBiquadFilter(); voice.type='bandpass';
  voice.frequency.value = 1200; voice.Q.value = 0.5;
  noiseGain.connect(voice); toneGain.connect(voice);
  voice.connect(master);
  noiseSrc = makeNoise();
  noiseSrc.connect(noiseGain);
  noiseSrc.start();
  STATIONS.forEach(s => s.live = s.build());
}

function retune(){
  if (!powered || !ctx) return;
  let bestSig = 0, bestName = null;
  STATIONS.forEach(s => {
    const d = Math.abs(freq - s.at);
    const sig = Math.max(0, 1 - d / s.w);        // 0..1
    const smooth = sig*sig*(3-2*sig);
    s.live.gain.gain.setTargetAtTime(smooth * 0.8, ctx.currentTime, 0.08);
    if (sig > bestSig){ bestSig = sig; bestName = s.name; }
  });
  noiseGain.gain.setTargetAtTime(0.12 * (1 - bestSig*0.92), ctx.currentTime, 0.08);
  const st = document.getElementById('station');
  st.textContent = bestSig > 0.32 ? bestName : `${freq.toFixed(1)} MHz — STATIC`;
  signalLevel = bestSig;
}
let signalLevel = 0;

// ---------------- UI ----------------
const needle = document.getElementById('needle');
function updateNeedle(){
  needle.style.left = ((freq - 88) / 20 * 100) + '%';
}
updateNeedle();

// rotary knob behaviour
function knobify(el, onDelta, initAngle=0){
  let angle = initAngle;
  el.style.transform = `rotate(${angle}deg)`;
  let dragging = false, lastA = 0;
  const center = () => {
    const r = el.getBoundingClientRect();
    return { x: r.left + r.width/2, y: r.top + r.height/2 };
  };
  el.addEventListener('pointerdown', e => {
    dragging = true;
    const c = center();
    lastA = Math.atan2(e.clientY - c.y, e.clientX - c.x);
    el.setPointerCapture(e.pointerId);
  });
  el.addEventListener('pointermove', e => {
    if (!dragging) return;
    const c = center();
    const a = Math.atan2(e.clientY - c.y, e.clientX - c.x);
    let d = a - lastA;
    if (d > Math.PI) d -= Math.PI*2;
    if (d < -Math.PI) d += Math.PI*2;
    lastA = a;
    angle += d * 180/Math.PI;
    el.style.transform = `rotate(${angle}deg)`;
    onDelta(d * 180/Math.PI);
  });
  el.addEventListener('pointerup', () => dragging = false);
}

knobify(document.getElementById('tuneKnob'), d => {
  freq = Math.max(88, Math.min(108, freq + d * 0.02));
  updateNeedle(); retune();
}, -80);

knobify(document.getElementById('volKnob'), d => {
  volume = Math.max(0, Math.min(1, volume + d * 0.004));
  if (master) master.gain.setTargetAtTime(volume, ctx.currentTime, 0.05);
}, 30);

document.getElementById('powerToggle').addEventListener('click', function(){
  powered = !powered;
  this.classList.toggle('on', powered);
  if (powered){
    if (!ctx) initAudio();
    ctx.resume();
    master.gain.setTargetAtTime(volume, ctx.currentTime, 0.3);
    retune();
  } else {
    master.gain.setTargetAtTime(0, ctx.currentTime, 0.15);
    document.getElementById('station').textContent = '— POWER OFF —';
    STATIONS.forEach(s => s.live && s.live.gain.gain.setTargetAtTime(0, ctx.currentTime, 0.1));
    noiseGain.gain.setTargetAtTime(0, ctx.currentTime, 0.1);
    signalLevel = 0;
  }
});

// VU needle wobble
const vuN = document.getElementById('vuNeedle');
(function vu(){
  let target = powered ? (-42 + signalLevel*70 + Math.random()*8*signalLevel + (powered?4:0)) : -42;
  vuN.style.transform = `rotate(${target}deg)`;
  setTimeout(vu, 90);
})();
</script>
