<script>
const clamp01 = v => Math.max(0, Math.min(1, v));

/* timecode */
setInterval(() => {
  const n = new Date();
  const f = String(Math.floor(n.getMilliseconds()/41.7)).padStart(2,'0');
  document.getElementById('tc').textContent =
    `${String(n.getHours()).padStart(2,'0')}:${String(n.getMinutes()).padStart(2,'0')}:${String(n.getSeconds()).padStart(2,'0')}:${f}`;
}, 41);

/* ---------- wipe deck ---------- */
const dtrack = document.getElementById('dtrack');
const slides = [...document.querySelectorAll('.slide')];
const drail = document.getElementById('drail');
const NAMES = ['ORIGINAL FILMS','COMMERCIAL WORK','MUSIC VIDEO','THE GRADE'];
slides.forEach(() => {
  const s = document.createElement('div');
  s.className = 'dseg'; s.innerHTML = '<i></i>';
  drail.appendChild(s);
});
const segs = [...drail.querySelectorAll('.dseg i')];
const dlabel = document.getElementById('deckLabel');
let curDeck = -1;

function deck(){
  const r = dtrack.getBoundingClientRect();
  const p = clamp01(-r.top / (r.height - innerHeight));
  const n = slides.length;
  const f = p * n;
  const idx = Math.min(n-1, Math.floor(f));
  const local = f - idx;

  slides.forEach((s, i) => {
    const clip = s.querySelector('.clipper');
    const im = s.querySelector('.sim');
    if (i < idx){ s.style.visibility='visible'; s.style.zIndex = i; clip.style.clipPath = 'inset(0)'; }
    else if (i === idx){
      s.style.visibility='visible'; s.style.zIndex = i;
      clip.style.clipPath = 'inset(0)';
      im.style.transform = `scale(${1.02 + local*0.06}) translateX(${local*-1.4}%)`;
    } else if (i === idx+1){
      // next slide wipes in from the right over the last 22% of current
      const w = local > 0.78 ? (local-0.78)/0.22 : 0;
      s.style.visibility = w > 0 ? 'visible' : 'hidden';
      s.style.zIndex = i;
      clip.style.clipPath = `inset(0 0 0 ${100 - w*100}%)`;
      im.style.transform = 'scale(1.05)';
    } else {
      s.style.visibility='hidden';
    }
    s.classList.toggle('live', i === idx && local < 0.86);
  });

  segs.forEach((seg, i) => {
    seg.style.width = i < idx ? '100%' : i === idx ? (local*100)+'%' : '0%';
  });
  if (idx !== curDeck){
    curDeck = idx;
    dlabel.textContent = `0${idx+1} / 0${slides.length} — ${NAMES[idx]}`;
  }
}
addEventListener('scroll', deck, { passive:true });
deck();

/* reveals */
const io = new IntersectionObserver(es => es.forEach(e => {
  if (e.isIntersecting) e.target.classList.add('in');
}), { threshold:0.16 });
document.querySelectorAll('.reveal').forEach((el,i) => {
  el.style.transitionDelay = (i%3)*0.1+'s';
  io.observe(el);
});

/* cursor-reactive ambient spotlight (signature reference glow) */
{
  const spot = document.getElementById('cursorSpot');
  addEventListener('pointermove', e => {
    document.body.classList.add('has-mouse');
    spot.style.setProperty('--sx', (e.clientX/innerWidth*100)+'%');
    spot.style.setProperty('--sy', (e.clientY/innerHeight*100)+'%');
  });
}

</script>
