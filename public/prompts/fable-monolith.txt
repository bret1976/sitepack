<script type="module">
import * as THREE from 'three';
import { RoomEnvironment } from 'three/addons/environments/RoomEnvironment.js';
import { EffectComposer } from 'three/addons/postprocessing/EffectComposer.js';
import { RenderPass } from 'three/addons/postprocessing/RenderPass.js';
import { UnrealBloomPass } from 'three/addons/postprocessing/UnrealBloomPass.js';

const container = document.getElementById('scene');
const renderer = new THREE.WebGLRenderer({ antialias: true });
renderer.setPixelRatio(Math.min(devicePixelRatio, 2));
renderer.setSize(innerWidth, innerHeight);
renderer.toneMapping = THREE.ACESFilmicToneMapping;
renderer.toneMappingExposure = 1.7;
container.appendChild(renderer.domElement);

const scene = new THREE.Scene();
scene.background = new THREE.Color(0x0a0a0b);
scene.fog = new THREE.FogExp2(0x0a0a0b, 0.055);

const camera = new THREE.PerspectiveCamera(42, innerWidth/innerHeight, 0.1, 100);
camera.position.set(0, 0.4, 7.2);

const pmrem = new THREE.PMREMGenerator(renderer);
scene.environment = pmrem.fromScene(new RoomEnvironment(), 0.04).texture;

// ---- the crystal ----
const geo = new THREE.IcosahedronGeometry(1.55, 0);
const mat = new THREE.MeshPhysicalMaterial({
  color: 0xffffff,
  metalness: 0,
  roughness: 0.02,
  transmission: 1,
  thickness: 1.6,
  ior: 1.7,
  dispersion: 4.0,
  envMapIntensity: 2.0,
  clearcoat: 1,
  clearcoatRoughness: 0.06,
  iridescence: 0.9,
  iridescenceIOR: 1.6,
  iridescenceThicknessRange: [120, 500],
  attenuationColor: new THREE.Color(0xbcd4ff),
  attenuationDistance: 6.5,
});
const crystal = new THREE.Mesh(geo, mat);
scene.add(crystal);

// inner core — faint emissive nucleus
const core = new THREE.Mesh(
  new THREE.IcosahedronGeometry(0.42, 1),
  new THREE.MeshBasicMaterial({ color: 0xcfe0ff, transparent: true, opacity: 0.9 })
);
scene.add(core);
const coreLight = new THREE.PointLight(0xaac8ff, 30, 12);
scene.add(coreLight);

// wireframe cage slightly larger, barely visible
const cage = new THREE.LineSegments(
  new THREE.EdgesGeometry(new THREE.IcosahedronGeometry(2.35, 0)),
  new THREE.LineBasicMaterial({ color: 0xe8e6e1, transparent: true, opacity: 0.07 })
);
scene.add(cage);

// ---- dust motes ----
const dustCount = 700;
const dustGeo = new THREE.BufferGeometry();
const pos = new Float32Array(dustCount * 3);
const phase = new Float32Array(dustCount);
for (let i = 0; i < dustCount; i++) {
  const r = 3 + Math.random() * 9;
  const th = Math.random() * Math.PI * 2;
  const ph = Math.acos(2*Math.random()-1);
  pos[i*3]   = r * Math.sin(ph) * Math.cos(th);
  pos[i*3+1] = (Math.random()-0.5) * 9;
  pos[i*3+2] = r * Math.sin(ph) * Math.sin(th);
  phase[i] = Math.random() * Math.PI * 2;
}
dustGeo.setAttribute('position', new THREE.BufferAttribute(pos, 3));
const dust = new THREE.Points(dustGeo, new THREE.PointsMaterial({
  color: 0xcfd6e4, size: 0.02, transparent: true, opacity: 0.55,
  sizeAttenuation: true, depthWrite: false,
}));
scene.add(dust);

// ---- lights ----
const key = new THREE.SpotLight(0xffffff, 420, 40, 0.6, 0.6);
key.position.set(6, 9, 4);
scene.add(key);
const rim = new THREE.PointLight(0x6f8fff, 90, 30);
rim.position.set(-7, -3, -5);
scene.add(rim);
const warm = new THREE.PointLight(0xffb98a, 45, 25);
warm.position.set(4, -5, 6);
scene.add(warm);

// pedestal ring of light on "floor"
const ring = new THREE.Mesh(
  new THREE.RingGeometry(2.2, 2.26, 128),
  new THREE.MeshBasicMaterial({ color: 0xe8e6e1, transparent: true, opacity: 0.22, side: THREE.DoubleSide })
);
ring.rotation.x = -Math.PI/2;
ring.position.y = -2.6;
scene.add(ring);

// ---- post ----
const composer = new EffectComposer(renderer);
composer.addPass(new RenderPass(scene, camera));
const bloom = new UnrealBloomPass(new THREE.Vector2(innerWidth, innerHeight), 0.35, 0.9, 0.82);
composer.addPass(bloom);

// ---- interaction: inertia drag orbit ----
let targetRotY = 0, targetRotX = 0;
let curRotY = 0, curRotX = 0;
let dragging = false, px = 0, py = 0;
let autoSpin = 0.0016;

const el = renderer.domElement;
el.style.touchAction = 'none';
el.addEventListener('pointerdown', e => { dragging = true; px = e.clientX; py = e.clientY; });
addEventListener('pointerup', () => dragging = false);
addEventListener('pointermove', e => {
  if (!dragging) return;
  targetRotY += (e.clientX - px) * 0.005;
  targetRotX += (e.clientY - py) * 0.003;
  targetRotX = Math.max(-0.9, Math.min(0.9, targetRotX));
  px = e.clientX; py = e.clientY;
});

const clock = new THREE.Clock();
function animate(){
  const t = clock.getElapsedTime();
  if (!dragging) targetRotY += autoSpin;

  curRotY += (targetRotY - curRotY) * 0.06;
  curRotX += (targetRotX - curRotX) * 0.06;

  // camera orbits; crystal tumbles slowly on its own axes
  const rad = 7.2;
  camera.position.x = Math.sin(curRotY) * rad * Math.cos(curRotX);
  camera.position.z = Math.cos(curRotY) * rad * Math.cos(curRotX);
  camera.position.y = 0.4 + Math.sin(curRotX) * rad * 0.55;
  camera.lookAt(0, 0, 0);

  crystal.rotation.x = t * 0.12;
  crystal.rotation.y = t * 0.17;
  crystal.position.y = Math.sin(t * 0.7) * 0.18;
  core.position.y = crystal.position.y;
  core.rotation.y = -t * 0.4;
  core.scale.setScalar(1 + Math.sin(t * 1.8) * 0.12);
  coreLight.position.y = crystal.position.y;
  coreLight.intensity = 26 + Math.sin(t * 1.8) * 10;
  cage.rotation.y = -t * 0.05;
  cage.rotation.z = t * 0.03;
  cage.position.y = crystal.position.y * 0.6;

  dust.rotation.y = t * 0.012;
  const dp = dust.geometry.attributes.position;
  for (let i = 0; i < dustCount; i += 7) {
    dp.array[i*3+1] += Math.sin(t*0.8 + phase[i]) * 0.0007;
  }
  dp.needsUpdate = true;

  composer.render();
  requestAnimationFrame(animate);
}
animate();
document.getElementById('loading').classList.add('done');

addEventListener('resize', () => {
  camera.aspect = innerWidth/innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(innerWidth, innerHeight);
  composer.setSize(innerWidth, innerHeight);
});
</script>
</body>
