<script>
const clamp01 = v => Math.max(0, Math.min(1, v));

/* world image parallax */
const worlds = [...document.querySelectorAll('.world')];
function par(){
  worlds.forEach(w => {
    const r = w.getBoundingClientRect();
    if (r.bottom < 0 || r.top > innerHeight) return;
    const p = (innerHeight - r.top) / (innerHeight + r.height); // 0..1
    w.querySelector('.wimg').style.transform = `translateY(${(p-0.5)*22}%) scale(1.18)`;
  });
  requestAnimationFrame(par);
}
requestAnimationFrame(par);

/* statement word illumination */
{
  const el = document.getElementById('stmt');
  el.innerHTML = el.textContent.split(' ').map(w => `<span class="sword">${w}</span>`).join(' ');
  const ws = [...el.querySelectorAll('.sword')];
  addEventListener('scroll', () => {
    const r = el.getBoundingClientRect();
    const p = clamp01((innerHeight*0.8 - r.top) / (innerHeight*0.85));
    const lit = Math.floor(p * ws.length * 1.12);
    ws.forEach((s,i) => s.classList.toggle('lit', i < lit));
  }, { passive:true });
}

/* reveals */
const io = new IntersectionObserver(es => es.forEach(e => {
  if (e.isIntersecting) e.target.classList.add('in');
}), { threshold:0.18 });
document.querySelectorAll('.reveal').forEach((el,i) => {
  el.style.transitionDelay = (i%3)*0.12+'s';
  io.observe(el);
});

/* cursor-reactive ambient spotlight (signature reference glow) */
{
  const spot = document.getElementById('cursorSpot');
  addEventListener('pointermove', e => {
    document.body.classList.add('has-mouse');
    spot.style.setProperty('--sx', (e.clientX/innerWidth*100)+'%');
    spot.style.setProperty('--sy', (e.clientY/innerHeight*100)+'%');
  });
}

</script>
