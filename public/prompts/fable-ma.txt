<script>
// enso draws as it scrolls into view
const enso = document.getElementById('ensoPath');
const len = enso.getTotalLength();
enso.style.strokeDasharray = len;
enso.style.strokeDashoffset = len;
addEventListener('scroll', () => {
  const r = enso.closest('section').getBoundingClientRect();
  const p = Math.max(0, Math.min(1, (innerHeight - r.top) / (innerHeight * 1.1)));
  enso.style.strokeDashoffset = len * (1 - p);
}, { passive:true });

// reveals
const io = new IntersectionObserver(es => es.forEach(e => {
  if (e.isIntersecting) e.target.classList.add('in');
}), { threshold:0.3 });
document.querySelectorAll('.reveal').forEach(el => io.observe(el));

// active nav
const secs = ['haru','natsu','aki','fuyu'].map(id => document.getElementById(id));
const links = document.querySelectorAll('.side-nav a');
const io2 = new IntersectionObserver(es => es.forEach(e => {
  if (e.isIntersecting){
    links.forEach((l,i) => l.classList.toggle('on', secs[i] === e.target));
  }
}), { threshold:0.5 });
secs.forEach(s => io2.observe(s));

// seasonal drift particles — change by scroll position (season)
const cv = document.getElementById('drift');
const cx = cv.getContext('2d');
let W, H;
function rs(){ W = cv.width = innerWidth; H = cv.height = innerHeight; }
addEventListener('resize', rs); rs();

const flakes = Array.from({length: 26}, () => ({
  x: Math.random()*1, y: Math.random()*1,
  s: 0.5 + Math.random(), ph: Math.random()*7,
  spin: Math.random()*7,
}));

const SEASONS = [
  { col:'rgba(216,150,160,0.7)', shape:'petal' },  // spring
  { col:'rgba(110,140,100,0.55)', shape:'leaf' },  // summer
  { col:'rgba(190,120,60,0.6)', shape:'leaf' },    // autumn
  { col:'rgba(255,255,255,0.9)', shape:'snow' },   // winter
];

function seasonIndex(){
  const total = document.body.scrollHeight - innerHeight;
  const p = scrollY / Math.max(1,total);
  if (p < 0.22) return -1;              // cover: nothing falls
  return Math.min(3, Math.floor((p - 0.22) / 0.195));
}

function draw(now){
  const t = now/1000;
  cx.clearRect(0,0,W,H);
  const si = seasonIndex();
  if (si >= 0){
    const s = SEASONS[si];
    cx.fillStyle = s.col;
    for (const f of flakes){
      const x = ((f.x + Math.sin(t*0.3 + f.ph)*0.05) % 1) * W;
      const y = ((f.y + t*0.018*f.s) % 1.1) * H;
      cx.save();
      cx.translate(x, y);
      cx.rotate(t*f.spin*0.2 + f.ph);
      if (s.shape === 'petal'){
        cx.beginPath();
        cx.ellipse(0, 0, 5*f.s, 3*f.s, 0, 0, 7);
        cx.fill();
      } else if (s.shape === 'leaf'){
        cx.beginPath();
        cx.moveTo(0,-5*f.s); cx.quadraticCurveTo(5*f.s,0, 0,5*f.s);
        cx.quadraticCurveTo(-5*f.s,0, 0,-5*f.s);
        cx.fill();
      } else {
        cx.beginPath();
        cx.arc(0,0, 2.2*f.s, 0, 7);
        cx.fill();
      }
      cx.restore();
    }
  }
  requestAnimationFrame(draw);
}
requestAnimationFrame(draw);
</script>
