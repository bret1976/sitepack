/* sticky study: swap fig when chapter crosses center */
const chapters = [...document.querySelectorAll('.chapter')];
const figs = [...document.querySelectorAll('.study-img')];
const cap = document.getElementById('studyCap');
let curFig = 0;
const io2 = new IntersectionObserver(es => es.forEach(e => {
  if (!e.isIntersecting) return;
  const i = +e.target.dataset.fig;
  if (i !== curFig){
    curFig = i;
    figs.forEach((f,j) => f.classList.toggle('on', j===i));
    cap.textContent = e.target.dataset.cap;
  }
}), { threshold:0.55 });
chapters.forEach(c => io2.observe(c));

/* pause parallax */
const pimg = document.querySelector('.pimg');
addEventListener('scroll', () => {
  const r = pimg.parentElement.getBoundingClientRect();
  if (r.bottom < 0 || r.top > innerHeight) return;
  const p = (innerHeight - r.top) / (innerHeight + r.height);
  pimg.style.transform = `translateY(${(p-0.5)*26}%) scale(1.15)`;
}, { passive:true });

/* reveals */
const io = new IntersectionObserver(es => es.forEach(e => {

}), { threshold:0.16 });

  el.style.transitionDelay = (i%3)*0.1+'s';
  io.observe(el);
