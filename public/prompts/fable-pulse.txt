const CORAL='#ff6b6b', MINT='#4ecdc4', GOLD='#ffd166', VIOLET='#a78bfa', MUTE='#5d6678';

function setup(id){
  const c = document.getElementById(id);
  const dpr = Math.min(devicePixelRatio, 2);
  const w = c.clientWidth, h = c.getAttribute('height')*1;
  c.width = w*dpr; c.height = h*dpr;
  c.style.height = h+'px';
  const x = c.getContext('2d');
  x.scale(dpr, dpr);
  return { x, w, h };
}

let charts = [];
function register(fn){ charts.push(fn); }

// ---- 1. heartbeat line ----
{
  let env;
  const heartVal = document.getElementById('heartVal');
  register(t => {
    if (!env) env = setup('heart');
    const {x,w,h} = env;
    x.clearRect(0,0,w,h);
    // grid
    x.strokeStyle = 'rgba(94,102,120,0.15)';
    x.lineWidth = 1;
    for (let i=1;i<4;i++){ x.beginPath(); x.moveTo(0,h*i/4); x.lineTo(w,h*i/4); x.stroke(); }
    // two waves: today vs yesterday
    const wave = (off, amp, col, fill) => {
      x.beginPath();
      for (let px=0; px<=w; px+=3){
        const u = px/w;
        // double-hump day curve + noise + traveling pulse
        const day = Math.exp(-Math.pow((u-0.35)*4.2,2))*0.7 + Math.exp(-Math.pow((u-0.74)*5,2))*0.95;
        const noise = Math.sin(u*40+t*2+off)*0.03 + Math.sin(u*13-t+off)*0.04;
        const pulse = Math.exp(-Math.pow(((u - (t*0.08+off)%1.2)*14),2))*0.22;
        const v = (day+noise+pulse)*amp;
        const py = h - 14 - v*(h-40);
        px===0 ? x.moveTo(px,py) : x.lineTo(px,py);
      }
      if (fill){
        x.save();
        x.lineTo(w,h); x.lineTo(0,h); x.closePath();
        const g = x.createLinearGradient(0,0,0,h);
        g.addColorStop(0,'rgba(78,205,196,0.28)'); g.addColorStop(1,'rgba(78,205,196,0)');
        x.fillStyle = g; x.fill();
        x.restore();
        x.beginPath();
        for (let px=0; px<=w; px+=3){
          const u = px/w;
          const day = Math.exp(-Math.pow((u-0.35)*4.2,2))*0.7 + Math.exp(-Math.pow((u-0.74)*5,2))*0.95;
          const noise = Math.sin(u*40+t*2+off)*0.03 + Math.sin(u*13-t+off)*0.04;

          px===0 ? x.moveTo(px,py) : x.lineTo(px,py);

      x.setLineDash(fill?[]:[4,5]);

    heartVal.textContent = (412 + Math.round(Math.sin(t*1.3)*24 + Math.sin(t*4.7)*9)).toLocaleString() + ' /min';
  });
}

{

{
