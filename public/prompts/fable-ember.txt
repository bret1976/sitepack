const clamp01 = v => Math.max(0, Math.min(1, v));

/* ============ procedural basketball texture ============ */
function makeBallTextures(){
  const W = 2048, H = 1024;
  const cv = document.createElement('canvas'); cv.width = W; cv.height = H;
  const g = cv.getContext('2d');
  const bump = document.createElement('canvas'); bump.width = W; bump.height = H;
  const bg = bump.getContext('2d');

  // base leather
  const grad = g.createLinearGradient(0,0,0,H);
  grad.addColorStop(0, '#c8501e');
  grad.addColorStop(0.5, '#d95a22');
  grad.addColorStop(1, '#b8481c');
  g.fillStyle = grad; g.fillRect(0,0,W,H);
  bg.fillStyle = '#808080'; bg.fillRect(0,0,W,H);

  // pebbles
  let seed = 12345;
  const rnd = () => (seed = (seed*16807)%2147483647)/2147483647;
  for (let i=0;i<26000;i++){
    const x = rnd()*W, y = rnd()*H;
    const r = 1.1 + rnd()*2.1;
    const l = rnd();
    g.fillStyle = `rgba(${l>0.5?255:60},${l>0.5?120:20},${l>0.5?60:5},${0.1+rnd()*0.16})`;
    g.beginPath(); g.arc(x,y,r,0,7); g.fill();
    bg.fillStyle = `rgba(${l>0.5?230:90},${l>0.5?230:90},${l>0.5?230:90},0.5)`;
    bg.beginPath(); bg.arc(x,y,r,0,7); bg.fill();
  }

  // seams in equirect space
  function seam(fn){
    g.strokeStyle = '#1a0d06'; g.lineWidth = 13; g.lineCap='round';
    bg.strokeStyle = '#000'; bg.lineWidth = 15;
    [g,bg].forEach(ctx => {
      ctx.beginPath();
      for (let px=0; px<=W; px+=4){
        const y = fn(px/W);
        if (y === null){ ctx.stroke(); ctx.beginPath(); continue; }
        px===0 ? ctx.moveTo(px, y*H) : ctx.lineTo(px, y*H);
      }
      ctx.stroke();
    });
  }
  // equator
  seam(u => 0.5);
  // two meridians (vertical lines in equirect become u=const)
  [0.0, 0.25, 0.5, 0.75].forEach(u0 => {
    [g,bg].forEach(ctx => {
      ctx.strokeStyle = ctx===g ? '#1a0d06' : '#000';
      ctx.lineWidth = ctx===g ? 13 : 15;
      ctx.beginPath(); ctx.moveTo(u0*W, 0); ctx.lineTo(u0*W, H); ctx.stroke();
    });
  });
  // two curved channel seams (classic basketball arcs)
  seam(u => 0.5 + 0.31*Math.sin(u*Math.PI*2));
  seam(u => 0.5 - 0.31*Math.sin(u*Math.PI*2));

  const map = new THREE.CanvasTexture(cv);
  map.colorSpace = THREE.SRGBColorSpace;
  map.anisotropy = 8;
  const bumpT = new THREE.CanvasTexture(bump);
  return { map, bump: bumpT };
}

(function animate(){

  if (!dragging){
