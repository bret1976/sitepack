<script>
const pages = [...document.querySelectorAll('.page')];
const dots = document.getElementById('dots');
pages.forEach((_, i) => {
  const d = document.createElement('div');
  d.className = 'dot' + (i === 0 ? ' on' : '');
  d.onclick = () => go(i);
  dots.appendChild(d);
});
let cur = 0;
function go(i){
  cur = Math.max(0, Math.min(pages.length - 1, i));
  pages.forEach((p, j) => p.classList.toggle('on', j === cur));
  [...dots.children].forEach((d, j) => d.classList.toggle('on', j === cur));
  document.getElementById('prev').disabled = cur === 0;
  document.getElementById('next').disabled = cur === pages.length - 1;
  document.getElementById('next').textContent = cur === pages.length - 1 ? 'the end ♡' : 'turn the page →';
}
document.getElementById('prev').onclick = () => go(cur - 1);
document.getElementById('next').onclick = () => go(cur + 1);
addEventListener('keydown', e => {
  if (e.key === 'ArrowRight') go(cur + 1);
  if (e.key === 'ArrowLeft') go(cur - 1);
});
go(0);
</script>
