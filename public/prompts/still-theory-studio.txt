 * Not used by the public Still Theory Studio marketing site.

import { drizzle } from "drizzle-orm/d1";
import * as schema from "./schema";

type D1Like = {
  prepare: (query: string) => unknown;
};

function getCloudflareDb(): D1Like | null {
  try {
    // Dynamic access so standard Next.js builds (Vercel/Railway) don't require cloudflare:workers
    // eslint-disable-next-line @typescript-eslint/no-require-imports
    const workers = (globalThis as { Cloudflare?: { env?: { DB?: D1Like } } }).Cloudflare;
    return workers?.env?.DB ?? null;
  } catch {
    return null;
  }
}

export function getDb() {
  const DB = getCloudflareDb();
  if (!DB) {
    throw new Error(
      "Database binding is unavailable. This marketing site does not require a database.",
    );
  }
  return drizzle(DB as never, { schema });
}

import { access, cp, mkdir, rm } from "node:fs/promises";
import { resolve } from "node:path";
import type { Plugin } from "vite";

async function exists(path: string): Promise<boolean> {
  try {
    await access(path);
    return true;
  } catch (error) {
    if ((error as NodeJS.ErrnoException).code === "ENOENT") {
      return false;
    }
    throw error;
  }
}

// Packages Sites metadata and migrations after Vite finishes compiling.
export function sites(): Plugin {
  let root = process.cwd();

  return {
    name: "sites",
    apply: "build",
    configResolved(config) {
      root = config.root;
    },
    async closeBundle() {
      const outputDirectory = resolve(root, "dist", ".openai");
      const hostingConfig = resolve(root, ".openai", "hosting.json");
      const drizzleSource = resolve(root, "drizzle");

      await rm(outputDirectory, { recursive: true, force: true });
      await mkdir(outputDirectory, { recursive: true });

      if (await exists(hostingConfig)) {
        await cp(hostingConfig, resolve(outputDirectory, "hosting.json"));
      }
      if (await exists(drizzleSource)) {
        await cp(drizzleSource, resolve(outputDirectory, "drizzle"), {
          recursive: true,
        });
      }
    },
  };
}

import type { MetadataRoute } from "next";

const BASE = "https://still-theory-studio.vercel.app";

const paths = ["/", "/studio", "/work", "/backstage", "/legal/privacy", "/legal/terms"];

export default function sitemap(): MetadataRoute.Sitemap {
  const now = new Date();
  return paths.map((path, i) => ({
    url: `${BASE}${path === "/" ? "" : path}`,
    lastModified: now,
    changeFrequency: i === 0 ? "weekly" : "monthly",
    priority: i === 0 ? 1 : path.startsWith("/legal") ? 0.3 : 0.7,
  }));
}

import assert from "node:assert/strict";

async function render(path = "/") {
  const workerUrl = new URL("../dist/server/index.js", import.meta.url);
  workerUrl.searchParams.set("test", `${process.pid}-${Date.now()}-${path}`);
  const { default: worker } = await import(workerUrl.href);

  return worker.fetch(
    new Request(`http://localhost${path}`, {
      headers: { accept: "text/html" },
    }),
    {
      ASSETS: {
        fetch: async () => new Response("Not found", { status: 404 }),
      },
    },
    {
      waitUntil() {},
      passThroughOnException() {},
    },
  );
}

test("homepage brands Still Theory Studio and serves the film shell", async () => {
  const response = await render("/");
  assert.equal(response.status, 200);
  assert.match(response.headers.get("content-type") ?? "", /^text\/html\b/i);

  const html = await response.text();
  assert.match(html, /Still Theory Studio/i);
  assert.doesNotMatch(html, /Ellusion Creative|Elusive Creative|ellusion\b/i);
  assert.match(html, /still theory studio/i);
  assert.match(html, /Ideas begin on paper/i);
  assert.match(html, /hello@stilltheory\.studio/i);
  assert.match(html, /film\/frames|film-layer|STILL THEORY STUDIO/i);
});

test("work, studio and backstage routes render with brand name", async () => {
  for (const path of ["/work", "/studio", "/backstage"]) {
    const response = await render(path);
    assert.equal(response.status, 200, `${path} should return 200`);
    const html = await response.text();
    assert.match(html, /Still Theory Studio|still theory studio/i);
    assert.doesNotMatch(html, /Ellusion Creative|Elusive Creative/i);
  }
});
