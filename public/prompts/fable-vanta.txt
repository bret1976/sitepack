import * as THREE from 'three';

/* ============ particle globe with settlement arcs ============ */
const canvas = document.getElementById('globe');
const renderer = new THREE.WebGLRenderer({ canvas, antialias:true, alpha:true });
renderer.setPixelRatio(Math.min(devicePixelRatio, 2));
renderer.setSize(innerWidth, innerHeight);

const scene = new THREE.Scene();
const camera = new THREE.PerspectiveCamera(38, innerWidth/innerHeight, 0.1, 100);
camera.position.set(0, 0.4, 7);

const globeGroup = new THREE.Group();
scene.add(globeGroup);
// offset globe to the right on wide screens
function placeGlobe(){
  globeGroup.position.x = innerWidth > 900 ? 1.9 : 0;
  globeGroup.position.y = innerWidth > 900 ? 0 : 1.2;
  globeGroup.scale.setScalar(innerWidth > 900 ? 1 : 0.72);
}
placeGlobe();

/* dots on sphere via fibonacci lattice */
const R = 2.3;
const COUNT = 2600;
const pos = new Float32Array(COUNT*3);
const shade = new Float32Array(COUNT);
for (let i=0;i<COUNT;i++){
  const t = i/COUNT;
  const ga = i * 2.39996323;
  const lat = Math.acos(1 - 2*t);
  const x = Math.sin(lat)*Math.cos(ga), y = Math.cos(lat), z = Math.sin(lat)*Math.sin(ga);
  // fake continents: keep dots where layered sines say "land"
  const landish = Math.sin(3.1*x + 1.3) * Math.cos(2.4*y - 0.8) + Math.sin(2.2*z + 4.1) * 0.7
                + Math.sin(5.3*x*y) * 0.35;
  shade[i] = landish > 0.15 ? 1 : 0.22;
  pos[i*3] = x*R; pos[i*3+1] = y*R; pos[i*3+2] = z*R;
}
const geo = new THREE.BufferGeometry();
geo.setAttribute('position', new THREE.BufferAttribute(pos, 3));
geo.setAttribute('aShade', new THREE.BufferAttribute(shade, 1));
const mat = new THREE.ShaderMaterial({
  transparent:true,
  depthWrite:false,
  uniforms:{ uTime:{value:0} },
  vertexShader:`
    attribute float aShade;
    varying float vShade;
    varying vec3 vPos;
    void main(){
      vShade = aShade;
      vPos = position;
      vec4 mv = modelViewMatrix * vec4(position, 1.0);
      gl_PointSize = (aShade > 0.5 ? 2.6 : 1.6) * (300.0 / -mv.z) * 0.02;
      gl_Position = projectionMatrix * mv;
    }`,
  fragmentShader:`
    varying float vShade;
    varying vec3 vPos;
    uniform float uTime;
    void main(){
      float d = length(gl_PointCoord - 0.5);
      if (d > 0.5) discard;
      float tw = 0.75 + 0.25*sin(uTime*1.5 + vPos.x*8.0 + vPos.y*6.0);
      vec3 land = vec3(0.35, 0.75, 0.65);
      vec3 sea  = vec3(0.18, 0.3, 0.5);
      vec3 col = mix(sea, land, step(0.5, vShade));
      gl_FragColor = vec4(col, vShade * tw * (1.0 - d*1.4));
    }`
});
globeGroup.add(new THREE.Points(geo, mat));

/* atmosphere rim */
const rim = new THREE.Mesh(
  new THREE.SphereGeometry(R*1.02, 48, 48),
  new THREE.ShaderMaterial({
    transparent:true, side:THREE.BackSide, depthWrite:false,
    vertexShader:`varying vec3 vN; void main(){ vN = normalize(normalMatrix * normal); gl_Position = projectionMatrix * modelViewMatrix * vec4(position,1.0); }`,
    fragmentShader:`varying vec3 vN; void main(){
      float glow = pow(0.72 - dot(vN, vec3(0,0,-1.0)), 3.0);
      gl_FragColor = vec4(0.24, 0.5, 1.0, glow*0.55);
    }`
  })
);
globeGroup.add(rim);

/* settlement arcs */
const arcs = [];
function randPointOnSphere(){
  const u = Math.random(), v = Math.random();
  const th = 2*Math.PI*u, ph = Math.acos(2*v-1);
  return new THREE.Vector3(
    R*Math.sin(ph)*Math.cos(th), R*Math.cos(ph), R*Math.sin(ph)*Math.sin(th));
}
function spawnArc(){
  const a = randPointOnSphere(), b = randPointOnSphere();
  if (a.distanceTo(b) < 1.6) return;
  const mid = a.clone().add(b).multiplyScalar(0.5).normalize().multiplyScalar(R * (1.25 + Math.random()*0.35));
  const curve = new THREE.QuadraticBezierCurve3(a, mid, b);
  const pts = curve.getPoints(48);
  const g = new THREE.BufferGeometry().setFromPoints(pts);
  const m = new THREE.LineBasicMaterial({
    color: Math.random() < 0.5 ? 0x2ee6a8 : 0x3d7bff,
    transparent:true, opacity:0
  });
  const line = new THREE.Line(g, m);
  line.userData = { born: performance.now(), life: 3200 + Math.random()*1800, max: 48 };
  g.setDrawRange(0, 0);
  globeGroup.add(line);
  arcs.push(line);
  // endpoint pulse
  const pulse = new THREE.Mesh(
    new THREE.SphereGeometry(0.03, 8, 8),
    new THREE.MeshBasicMaterial({ color: m.color, transparent:true })
  );
  pulse.position.copy(b);
  pulse.userData = { born: performance.now(), life: 2000 };
  globeGroup.add(pulse);
  arcs.push(pulse);
}

let scrollN = 0;

const clock = new THREE.Clock();
function animate(){
  const t = clock.getElapsedTime();
  mat.uniforms.uTime.value = t;
  globeGroup.rotation.y = t*0.06 + scrollN*0.35;
  globeGroup.rotation.x = 0.25 + Math.sin(t*0.1)*0.04;
  // globe recedes as you scroll
  const fade = Math.max(0, 1 - scrollN*0.45);
  globeGroup.position.z = -scrollN * 1.2;
  canvas.style.opacity = 0.25 + fade*0.75;

  const now = performance.now();
  for (let i=arcs.length-1;i>=0;i--){
    const o = arcs[i];
    const age = now - o.userData.born;
    const p = age / o.userData.life;
    if (p >= 1){ globeGroup.remove(o); arcs.splice(i,1); continue; }
    if (o.isLine){
      const draw = p < 0.4 ? (p/0.4) : 1;
      o.geometry.setDrawRange(0, Math.floor(draw * o.userData.max));
      o.material.opacity = p < 0.4 ? 0.9 : 0.9 * (1 - (p-0.4)/0.6);
    } else {
      o.material.opacity = 1 - p;
      o.scale.setScalar(1 + p*2.4);
    }
  }
  renderer.render(scene, camera);
  requestAnimationFrame(animate);
}
