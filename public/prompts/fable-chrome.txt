// -------- metaballs with chrome-ish shading, 2D field rendered per-pixel at low res, upscaled --------
const cv = document.getElementById('metal');
const ctx2 = cv.getContext('2d');

// low-res field buffer for speed
let FW, FH, W, H, SCALE = 4;
let field, img;
function rs(){
  W = cv.width = innerWidth;
  H = cv.height = innerHeight;
  FW = Math.ceil(W/SCALE); FH = Math.ceil(H/SCALE);
  img = ctx2.createImageData(FW, FH);
}
addEventListener('resize', rs); rs();

let balls = [];
function spawn(n){
  balls = Array.from({length:n}, (_,i) => ({
    x: Math.random()*W, y: Math.random()*H,
    vx: (Math.random()-0.5)*2.2, vy: (Math.random()-0.5)*2.2,
    r: 40 + Math.random()*70,
  }));
}
spawn(7);

let magnet = true;
let mx = -9999, my = -9999;
addEventListener('pointermove', e => { mx = e.clientX; my = e.clientY; });
addEventListener('pointerleave', () => { mx = -9999; my = -9999; });

document.querySelectorAll('.pill-btn[data-n]').forEach(b => {
  b.onclick = () => {
    spawn(+b.dataset.n);
    document.querySelectorAll('.pill-btn[data-n]').forEach(x => x.classList.toggle('on', x===b));
  };
});
document.getElementById('magnet').onclick = function(){
  magnet = !magnet;
  this.textContent = 'Magnet: ' + (magnet?'On':'Off');
  this.classList.toggle('on', false);
};

function frame(t){
  // physics
  for (const b of balls){
    if (magnet && mx > -999){
      const dx = mx - b.x, dy = my - b.y;
      const d = Math.hypot(dx,dy)+1;
      const f = Math.min(0.35, 4200/(d*d));
      b.vx += dx/d * f; b.vy += dy/d * f;
    }
    b.x += b.vx; b.y += b.vy;
    b.vx *= 0.992; b.vy *= 0.992;
    if (b.x < b.r*0.3){ b.vx = Math.abs(b.vx); }
    if (b.x > W - b.r*0.3){ b.vx = -Math.abs(b.vx); }
    if (b.y < b.r*0.3){ b.vy = Math.abs(b.vy); }
    if (b.y > H - b.r*0.3){ b.vy = -Math.abs(b.vy); }
    // gentle idle wander
    b.vx += Math.sin(t*0.001 + b.r)*0.012;
    b.vy += Math.cos(t*0.0013 + b.r)*0.012;
  }

  // field + shading
  const d = img.data;
  let p = 0;
  for (let y=0; y<FH; y++){
    const py = y*SCALE;
    for (let x=0; x<FW; x++){
      const px = x*SCALE;
      // metaball field + gradient
      let v=0, gx=0, gy=0;
      for (const b of balls){
        const dx = px-b.x, dy = py-b.y;
        const q = dx*dx + dy*dy + 40;
        const c = b.r*b.r/q;
        v += c;
        gx += -2*c*dx/q; gy += -2*c*dy/q;
      }
      if (v > 0.85){
        // inside: chrome shading from field gradient as pseudo-normal
        const gl = Math.hypot(gx, gy) + 1e-5;
        let nx = gx/gl, ny = gy/gl;
        // edge factor: near threshold = rim
        const edge = Math.min(1, (v-0.85)*3.2);
        // classic chrome: bright sky above the horizon, deep ground below,
        // hard dark band at the horizon line, soft moving env highlight
        const skyMix = 0.5 - ny*0.5;                    // 1 at top, 0 at bottom
        const horizon = Math.exp(-Math.pow(ny*4.5, 2)); // dark band where normal is horizontal
        const gleam = Math.max(0, Math.sin(nx*1.8 + t*0.0009)) * 0.5;
        let r = 26 + skyMix*185 + gleam*60;
        let g2 = 36 + skyMix*200 + gleam*66;
        let bl = 64 + skyMix*225 + gleam*70;
        r *= (1-horizon*0.82); g2 *= (1-horizon*0.78); bl *= (1-horizon*0.6);
        // fresnel rim light
        const rim = Math.pow(1-edge, 2);
        r += rim*110; g2 += rim*150; bl += rim*190;
        d[p]=Math.min(255,r); d[p+1]=Math.min(255,g2); d[p+2]=Math.min(255,bl); d[p+3]=255;
      } else if (v > 0.6){
        // soft outer glow
        const a = (v-0.6)/0.25;
        d[p]=40*a; d[p+1]=70*a; d[p+2]=140*a; d[p+3]=Math.floor(a*130);
      } else {
        d[p+3]=0;
      }
      p += 4;
    }
  }

  ctx2.clearRect(0,0,W,H);
  // upscale blit with smoothing + slight blur to melt the pixel grid
  ctx2.imageSmoothingEnabled = true;
  const off = frame.off || (frame.off = document.createElement('canvas'));
  if (off.width !== FW || off.height !== FH){ off.width = FW; off.height = FH; }
  off.getContext('2d').putImageData(img, 0, 0);
  ctx2.filter = 'blur(2px) saturate(1.15)';
  ctx2.drawImage(off, 0, 0, W, H);
  ctx2.filter = 'none';

  requestAnimationFrame(frame);
}
requestAnimationFrame(frame);
</script>
