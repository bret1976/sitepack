/* ---------- cursor ---------- */
const cur = document.getElementById('cur');
let cx=-200, cy=-200, tx=cx, ty=cy;
cur.style.left = '-200px'; cur.style.top = '-200px';
addEventListener('pointermove', e => { tx = e.clientX; ty = e.clientY; });
(function curLoop(){
  cx += (tx-cx)*0.28; cy += (ty-cy)*0.28;
  cur.style.left = cx+'px'; cur.style.top = cy+'px';
  requestAnimationFrame(curLoop);
})();

/* ---------- clock ---------- */
setInterval(() => {
  document.getElementById('clock').textContent =
    new Date().toTimeString().slice(0,8) + ' AMS';
}, 1000);

/* ---------- kinetic hero rows ---------- */
function fillRow(id, words, cls){
  const row = document.getElementById(id);
  let html = '';
  for (let r=0;r<6;r++) words.forEach((w,i) => {
    html += `<span class="wd ${cls && i===cls.idx ? cls.cls : ''}">${w}</span>`;
  });
  row.innerHTML = html;
}
fillRow('kr1', ['MOTION','IS','MEANING','—'], {idx:2, cls:'org'});
fillRow('kr2', ['STILLNESS','IS','A','CHOICE','—'], null);
fillRow('kr3', ['MAKE','IT','MOVE','—'], {idx:2, cls:'org'});

/* velocity-reactive marquee: rows drift, scroll velocity skews + speeds them */
let vel = 0, lastScroll = scrollY;
addEventListener('scroll', () => {
  vel += (scrollY - lastScroll);
  lastScroll = scrollY;
}, { passive:true });

const klines = [...document.querySelectorAll('.kline')];
const offsets = klines.map(() => 0);
(function kinLoop(){
  vel *= 0.9;
  klines.forEach((kl, i) => {
    const speed = parseFloat(kl.dataset.speed);
    offsets[i] -= speed * (1.2 + Math.abs(vel)*0.06);
    const row = kl.firstElementChild;
    const half = row.scrollWidth / 2;
    if (offsets[i] < -half) offsets[i] += half;
    if (offsets[i] > 0) offsets[i] -= half;
    const skew = Math.max(-14, Math.min(14, vel * 0.22 * (speed > 0 ? 1 : -1)));
    kl.style.transform = `translateX(${offsets[i]}px) skewX(${skew}deg)`;
  });
  requestAnimationFrame(kinLoop);
})();

/* ---------- caps marquee duplicate ---------- */
const caps = document.getElementById('caps');
caps.innerHTML += caps.innerHTML;

/* ---------- work rows: cursor-following generative previews ---------- */
const peek = document.getElementById('peek');
const pcv = document.getElementById('peekCv');
const pg = pcv.getContext('2d');
let peekOn = false, peekHue = 20, peekSeed = 1;

document.querySelectorAll('.wrow').forEach((row, ri) => {
  row.addEventListener('pointerenter', () => {
    peekOn = true; peekHue = +row.dataset.hue; peekSeed = ri*97+13;
    peek.classList.add('on');
    document.body.classList.add('cur-view');
  });
  row.addEventListener('pointerleave', () => {
    peekOn = false;
    peek.classList.remove('on');
    document.body.classList.remove('cur-view');
  });
});
addEventListener('pointermove', e => {
  peek.style.left = e.clientX+'px';
  peek.style.top = e.clientY+'px';
});

/* generative "style frame" animation in the peek window */
(function peekLoop(now){
  if (peekOn){
    const t = now/1000;
    const W = 800, H = 600;
    pg.fillStyle = `hsl(${peekHue} 62% 12%)`;
    pg.fillRect(0,0,W,H);
    let seed = peekSeed;
    const rnd = () => (seed = (seed*16807)%2147483647)/2147483647;
    // sweeping bars
    for (let i=0;i<9;i++){
      const y = rnd()*H;
      const h = 14 + rnd()*70;
      const sp = 0.3 + rnd()*1.4;
      const x = ((t*sp*160 + rnd()*W) % (W+400)) - 200;
      pg.fillStyle = `hsla(${peekHue + rnd()*40-20} ${70+rnd()*30}% ${55+rnd()*25}% / ${0.5+rnd()*0.5})`;
      pg.fillRect(x, y, 120+rnd()*260, h);
    }
    // big type ghost
    pg.font = '900 190px Anton, sans-serif';
    pg.fillStyle = `hsla(${peekHue} 90% 88% / 0.9)`;
    pg.textAlign = 'center';
    const words = ['MOVE','PLAY','FAST','LOUD','LIVE'];
    pg.save();
    pg.translate(W/2, H/2 + 60);
    pg.scale(1 + Math.sin(t*2.4)*0.04, 1);
    pg.fillText(words[peekSeed % words.length], 0, 0);
    pg.restore();
    // scanline
    const sy = (t*260) % H;
    pg.fillStyle = 'rgba(255,255,255,0.14)';
    pg.fillRect(0, sy, W, 3);
  }
  requestAnimationFrame(peekLoop);
})(0);

/* ---------- reveals ---------- */
const io = new IntersectionObserver(es => es.forEach(e => {
  if (e.isIntersecting) e.target.classList.add('in');
}), { threshold:0.14 });
document.querySelectorAll('.reveal').forEach((el,i) => {
  el.style.transitionDelay = (i%3)*0.1+'s';
  io.observe(el);
});
