// split each [data-split] into word/letter spans
document.querySelectorAll('[data-split]').forEach(el => {
  const walk = node => {
    [...node.childNodes].forEach(child => {
      if (child.nodeType === 3) {
        const frag = document.createDocumentFragment();
        child.textContent.split(/(\s+)/).forEach(tok => {
          if (/^\s+$/.test(tok)) { frag.appendChild(document.createTextNode(tok)); return; }
          if (!tok) return;
          const span = document.createElement('span');
          span.className = 'w';
          span.textContent = tok;
          frag.appendChild(span);
        });
        node.replaceChild(frag, child);
      } else if (child.nodeType === 1 && child.tagName !== 'BR') {
        walk(child);
      }
    });
  };
  walk(el);
});

// title letters get individual spans
const title = document.querySelector('.title');
{
  const word = title.querySelector('.w');
  const letters = word.textContent.split('');
  word.textContent = '';
  letters.forEach(ch => {
    const s = document.createElement('span');
    s.className = 'w l';
    s.textContent = ch;
    word.appendChild(s);
  });
}

const stages = [...document.querySelectorAll('.stage')];
const bar = document.getElementById('bar');

function clamp(v, a, b){ return Math.max(a, Math.min(b, v)); }

// progress of a stage through the viewport: 0 = entering, 0.5 = centered, 1 = leaving
function stageP(el){
  const r = el.getBoundingClientRect();
  return clamp((innerHeight - r.top) / (innerHeight + r.height), 0, 1);
}

function apply(){
  bar.style.width = (scrollY / (document.body.scrollHeight - innerHeight) * 100) + '%';

  // title: variable weight breathes with scroll
  const tp = stageP(stages[0]);
  title.querySelectorAll('.l').forEach((l, i) => {
    const wgt = 100 + Math.sin((tp * 4 + i * 0.55) * Math.PI) * 400 + 400;
    l.style.fontVariationSettings = `'opsz' 144, 'wght' ${clamp(wgt, 100, 900)}`;
    l.style.transform = `translateY(${Math.sin(tp * 6 + i) * tp * -26}px)`;
    l.style.opacity = 1 - clamp((tp - 0.62) * 3.2, 0, 1);
  });

  stages.slice(1).forEach(stage => {
    const p = stageP(stage);
    const fx = stage.dataset.fx;
    const words = stage.querySelectorAll('.w');
    const n = words.length;
    words.forEach((w, i) => {
      // per-word local progress with stagger
      const lp = clamp((p - 0.12 - (i / n) * 0.14) * 3.2, 0, 1);
      const out = clamp((p - 0.78) * 5, 0, 1); // exit fade
      const ease = 1 - Math.pow(1 - lp, 3);
      let tf = '', op = ease * (1 - out), fil = '';

      if (fx === 'rise') {
        tf = `translateY(${(1-ease)*90}px) rotate(${(1-ease)*(i%2?4:-4)}deg)`;
        fil = `blur(${(1-ease)*10}px)`;
      } else if (fx === 'slide') {
        tf = `translateX(${(1-ease)*-140}px)`;
        fil = `blur(${(1-ease)*6}px)`;
      } else if (fx === 'weight') {
        const wg = 100 + ease * 800;
        w.style.fontVariationSettings = `'wght' ${w.classList.contains('heavy') ? clamp(wg,100,900) : 300}`;
        tf = `translateY(${(1-ease)*40}px)`;
      } else if (fx === 'scatter') {
        const cx = (p - 0.5) * 2; // -1..1 through center
        const mag = clamp(Math.abs(cx) * 2 - 0.35, 0, 1) * 160;
        const ang = (i * 137.5) % 360 * Math.PI / 180;
        tf = `translate(${Math.cos(ang)*mag}px, ${Math.sin(ang)*mag}px) rotate(${mag * (i%2?0.35:-0.35)}deg)`;
        op = ease * (1 - clamp(Math.abs(cx)*1.6 - 0.6, 0, 1));
        fil = `blur(${mag*0.05}px)`;
      } else if (fx === 'gather') {
        const ang = (i * 222.5) % 360 * Math.PI / 180;
        const d = (1-ease) * 200;
        tf = `translate(${Math.cos(ang)*d}px, ${Math.sin(ang)*d}px) scale(${0.6 + ease*0.4})`;
        fil = `blur(${(1-ease)*8}px)`;
      } else if (fx === 'wave') {
        const t = p * Math.PI * 2;
        tf = `translateY(${(1-ease)*70 + Math.sin(t*2 + i*0.8) * 14 * ease}px)`;
        fil = `blur(${(1-ease)*8}px)`;
      }
      w.style.transform = tf;
      w.style.opacity = op;
      if (fil) w.style.filter = fil;
    });
  });
  requestAnimationFrame(apply);
}
requestAnimationFrame(apply);
