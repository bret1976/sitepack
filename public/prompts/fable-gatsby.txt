// build sunburst rays
const burst = document.getElementById('burst');
const NS = 'http://www.w3.org/2000/svg';
for (let i = 0; i < 72; i++){
  const a = (i / 72) * Math.PI * 2;
  const line = document.createElementNS(NS, 'line');
  const r1 = 34, r2 = i % 2 ? 95 : 78;
  line.setAttribute('x1', 100 + Math.cos(a) * r1);
  line.setAttribute('y1', 100 + Math.sin(a) * r1);
  line.setAttribute('x2', 100 + Math.cos(a) * r2);
  line.setAttribute('y2', 100 + Math.sin(a) * r2);
  line.setAttribute('stroke-width', '0.45');
  burst.appendChild(line);
}
let rot = 0;
(function spin(){
  rot += 0.008;
  burst.style.transform = `translate(-50%,-50%) rotate(${rot}deg)`;
  requestAnimationFrame(spin);
})();

const io = new IntersectionObserver(es => es.forEach(e => {
  if (e.isIntersecting) e.target.classList.add('in');
}), { threshold:0.15 });
document.querySelectorAll('.reveal').forEach((el, i) => {
  el.style.transitionDelay = (i % 3) * 0.12 + 's';
  io.observe(el);
});
