const stage = document.getElementById('stage');

let bodies = [];
let gravityOn = false;

function spawn(){
  bodies.forEach(b => b.el.remove());
  bodies = [];
  SHAPES.forEach(s => {
    const el = document.createElement('div');
    el.className = 'shape';
    el.innerHTML = s.svg;
    stage.appendChild(el);
    const w = el.firstChild.getAttribute('width') * 1;
    const h = el.firstChild.getAttribute('height') * 1;
    const b = {
      el, w, h,
      x: s.x * innerWidth - w/2, y: s.y * innerHeight - h/2,
      vx: 0, vy: 0,
      rot: s.rot || 0, vr: 0,
      dragging: false,
    };
    bodies.push(b);

    el.addEventListener('pointerdown', e => {
      b.dragging = true;
      b.ox = e.clientX - b.x; b.oy = e.clientY - b.y;
      b.px = e.clientX; b.py = e.clientY;
      el.setPointerCapture(e.pointerId);
    });
    el.addEventListener('pointermove', e => {
      if (!b.dragging) return;
      b.x = e.clientX - b.ox;
      b.y = e.clientY - b.oy;
      b.vx = (e.clientX - b.px) * 0.9;
      b.vy = (e.clientY - b.py) * 0.9;
      b.vr = b.vx * 0.18;
      b.px = e.clientX; b.py = e.clientY;
    });
    el.addEventListener('pointerup', () => b.dragging = false);
  });
}
spawn();

function step(){
  for (const b of bodies){
    if (!b.dragging){
      if (gravityOn) b.vy += 0.55;
      b.x += b.vx; b.y += b.vy;
      b.rot += b.vr;
      b.vx *= 0.995; b.vy *= 0.995; b.vr *= 0.99;

      // walls
      if (b.x < 0){ b.x = 0; b.vx = -b.vx * 0.75; b.vr = -b.vr * 0.6; }
      if (b.x + b.w > innerWidth){ b.x = innerWidth - b.w; b.vx = -b.vx * 0.75; b.vr = -b.vr*0.6; }
      if (b.y < 0){ b.y = 0; b.vy = -b.vy * 0.75; }
      if (b.y + b.h > innerHeight){
        b.y = innerHeight - b.h;
        b.vy = -b.vy * 0.68;
        b.vx *= 0.92;
        if (Math.abs(b.vy) < 0.8) b.vy = 0;
      }
      // gentle idle float when gravity off
      if (!gravityOn && Math.abs(b.vx) < 0.05 && Math.abs(b.vy) < 0.05){
        b.y += Math.sin(performance.now()/900 + b.x) * 0.12;
      }
    }
    b.el.style.transform = `translate(${b.x}px, ${b.y}px) rotate(${b.rot}deg)`;
  }
  // pairwise elastic-ish collision (circle approx)
  for (let i = 0; i < bodies.length; i++){
    for (let j = i+1; j < bodies.length; j++){
      const a = bodies[i], c = bodies[j];
      const ax = a.x + a.w/2, ay = a.y + a.h/2;
      const cx2 = c.x + c.w/2, cy2 = c.y + c.h/2;
      const ra = Math.min(a.w, a.h)/2, rc = Math.min(c.w, c.h)/2;
      const dx = cx2 - ax, dy = cy2 - ay;
      const d = Math.hypot(dx, dy) || 1;
      const overlap = ra + rc - d;
      if (overlap > 0){
        const nx = dx/d, ny = dy/d;
        const push = overlap/2;
        if (!a.dragging){ a.x -= nx*push; a.y -= ny*push; }
        if (!c.dragging){ c.x += nx*push; c.y += ny*push; }
        const rel = (a.vx - c.vx)*nx + (a.vy - c.vy)*ny;
        if (rel > 0){
          const imp = rel * 0.82;
          if (!a.dragging){ a.vx -= imp*nx; a.vy -= imp*ny; a.vr += imp*0.4; }
          if (!c.dragging){ c.vx += imp*nx; c.vy += imp*ny; c.vr -= imp*0.4; }
        }
      }
    }
  }
  requestAnimationFrame(step);
}
