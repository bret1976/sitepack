const routeOrder = ["about", "dcrbn", "speaking", "legacy"];

function routeFromHash() {
  const route = window.location.hash.replace(/^#/, "");
  return pages[route] ? route : "home";
}

function setRoute(route, push = true) {
  const normalized = pages[route] ? route : "home";
  if (push) {
    const hash = normalized === "home" ? "#home" : `#${normalized}`;
    if (window.location.hash !== hash) history.pushState(null, "", hash);
  }
  renderRoute(normalized);
}

function openSidebar() {
  if (!runtime.size) {
    window.setTimeout(openSidebar, 100);
    return;
  }
  appState.isSidebarActive.value = true;
}

function closeSidebar() {
  appState.isSidebarActive.value = false;
}

function renderRoute(route) {
  document.querySelectorAll("[data-mz-route]").forEach((link) => {
    link.classList.toggle("is-active", link.dataset.mzRoute === route);
  });

  if (route === "home") {
    closeSidebar();
    return;
  }

  const page = pages[route] || pages.about;
  const title = document.querySelector(".js-mz-page-title");
  const body = document.querySelector(".js-mz-page-body");
  if (title) title.textContent = page.title;
  if (body) body.innerHTML = page.body;
  openSidebar();
}

function wireNavigation() {
  document.addEventListener("click", (event) => {
    const routeLink = event.target.closest("[data-mz-route]");
    if (!routeLink) return;
    event.preventDefault();
    setRoute(routeLink.dataset.mzRoute);
  });

  document.querySelector(".js-learn")?.addEventListener("click", () => {
    if (routeFromHash() === "home") setRoute("about");
    else renderRoute(routeFromHash());
  });

  window.addEventListener("hashchange", () => renderRoute(routeFromHash()));
  window.addEventListener("popstate", () => renderRoute(routeFromHash()));
  renderRoute(routeFromHash());
}

function boot() {
  const sidebar = document.querySelector(".js-sidebar");
  if (!sidebar) return;

  sidebar.innerHTML = `
    <nav class="mz-nav" aria-label="Majid Zafer sections">
      <a href="#home" data-mz-route="home">Home</a>
      ${routeOrder.map((route) => `<a href="#${route}" data-mz-route="${route}">${route}</a>`).join("")}
    </nav>
    <div class="font-bold uppercase mb-5 mz-page-title js-mz-page-title" data-astro-cid-edwrpnhe></div>
    <div class="mz-page-body js-mz-page-body" data-astro-cid-edwrpnhe></div>
    <div class="w-full h-[1px] bg-white my-20 opacity-50" data-astro-cid-edwrpnhe></div>
    <a href="#home" class="mz-home-link" data-mz-route="home" data-astro-cid-edwrpnhe>Back home</a>
    <div class="font-bold pointer-events-auto cursor-pointer js-sidebar-close" data-astro-cid-edwrpnhe>CLOSE</div>
  `;

  document.querySelector(".js-sidebar-close")?.addEventListener("click", () => {
    closeSidebar();
    if (window.location.hash && window.location.hash !== "#home") history.pushState(null, "", "#home");
  });

  document.querySelector('a[href="https://majidzafer.is/"]')?.setAttribute("href", "#home");
  document.querySelector('a[href="#home"]')?.removeAttribute("target");
  wireNavigation();
}

if (document.readyState === "loading") {
  document.addEventListener("DOMContentLoaded", boot);
} else {
  boot();
}
